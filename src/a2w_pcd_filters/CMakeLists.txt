cmake_minimum_required(VERSION 3.8)
project(a2w_pcd_filters)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters search octree)
find_package(OpenMP REQUIRED COMPONENTS CXX)
add_library(pcd_filters STATIC src/parallel_filters.cpp)
target_compile_features(pcd_filters PUBLIC cxx_std_17)
target_include_directories(pcd_filters PUBLIC include ${PCL_INCLUDE_DIRS})
target_link_libraries(pcd_filters PUBLIC ${PCL_LIBRARIES} OpenMP::OpenMP_CXX)
add_executable(pcd_ground_filter src/pcd_ground_filter.cpp)
target_link_libraries(pcd_ground_filter pcd_filters)
install(TARGETS pcd_ground_filter DESTINATION lib/${PROJECT_NAME})
install(FILES UPSTREAM.md src/parallel_filters.cpp DESTINATION share/${PROJECT_NAME}/licenses)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_parallel_filters test/test_parallel_filters.cpp)
  target_link_libraries(test_parallel_filters pcd_filters)
endif()
ament_package()
